\begin{tikzpicture}[
    font=\small,
    node distance=0.9cm and 1.4cm,
    box/.style={draw, rounded corners=3pt, minimum width=1.6cm,
                minimum height=0.7cm, align=center, fill=#1!15},
    arr/.style={-{Stealth[length=4pt]}, thick},
    darr/.style={{Stealth[length=4pt]}-{Stealth[length=4pt]}, thick, dashed},
    lbl/.style={font=\tiny, midway, fill=white, inner sep=1pt}
]

%% ── UEs ──────────────────────────────────────────────────────────────────
\node[box=blue] (ue1) {UE 1};
\node[box=blue, below=0.35cm of ue1] (ue2) {UE 2};
\node[box=blue, below=0.35cm of ue2] (ue3) {UE $N$};
\node[fit=(ue1)(ue2)(ue3), draw, dotted, inner sep=4pt,
      label=above:{\tiny User Equipment}] (ue_grp) {};

%% ── gNBs ─────────────────────────────────────────────────────────────────
\node[box=green, right=1.6cm of ue1] (gnb1) {gNB 1\\{\tiny NR/THz}};
\node[box=green, right=1.6cm of ue3] (gnb2) {gNB $M$\\{\tiny NR/THz}};
\node[fit=(gnb1)(gnb2), draw, dotted, inner sep=4pt,
      label=above:{\tiny RAN Layer}] (gnb_grp) {};

%% ── MEC hosts ────────────────────────────────────────────────────────────
\node[box=orange, right=1.5cm of gnb1] (mec1) {MEC\\Host 1};
\node[box=orange, right=1.5cm of gnb2] (mec2) {MEC\\Host $M$};
\node[fit=(mec1)(mec2), draw, dotted, inner sep=4pt,
      label=above:{\tiny Edge Compute}] (mec_grp) {};

%% ── Near-RT RIC ──────────────────────────────────────────────────────────
\node[box=red, right=2.8cm of gnb1, yshift=-0.55cm] (ric) {Near-RT\\RIC};

%% ── xApps ────────────────────────────────────────────────────────────────
\node[box=violet, above right=0.3cm and 0.9cm of ric] (ho_xapp)
      {HO\\xApp\\{\tiny DQN}};
\node[box=violet, below right=0.3cm and 0.9cm of ric] (fl_xapp)
      {FL\\xApp\\{\tiny FedAvg}};

%% ── Digital Twin ─────────────────────────────────────────────────────────
\node[box=teal, below=0.8cm of ric] (twin) {Digital\\Twin};

%% ── Cloud ────────────────────────────────────────────────────────────────
\node[box=gray, right=1.6cm of ho_xapp, yshift=-0.55cm] (cloud)
      {Cloud / Non-RT\\RIC (A1)};

%% ── Arrows ───────────────────────────────────────────────────────────────
% UE <-> gNB  (air interface)
\draw[darr] (ue1.east) -- node[lbl]{NR/THz} (gnb1.west);
\draw[darr] (ue3.east) -- (gnb2.west);

% gNB -> near-RT RIC  (E2)
\draw[arr] (gnb1.east) -- node[lbl,above]{E2} (ric.west|-gnb1);
\draw[arr] (gnb2.east) -- (ric.west|-gnb2);

% RIC <-> xApps
\draw[darr] (ric.north east) -- (ho_xapp.south west);
\draw[darr] (ric.south east) -- (fl_xapp.north west);

% RIC <-> Digital Twin
\draw[darr] (ric.south) -- node[lbl,right]{KPI} (twin.north);

% gNB <-> MEC
\draw[darr] (gnb1.east|-mec1) -- node[lbl]{MEC API} (mec1.west);
\draw[darr] (gnb2.east|-mec2) -- (mec2.west);

% xApps -> Cloud
\draw[arr] (ho_xapp.east) -- (cloud.west|-ho_xapp);
\draw[arr] (fl_xapp.east) -- (cloud.west|-fl_xapp);

\end{tikzpicture}
